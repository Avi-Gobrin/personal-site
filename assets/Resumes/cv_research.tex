\section{Research and Projects}
\cvSubHeadingListStart

  \cvProjectHeading
    {{\bf Latent-Structure Modeling of Student Outcomes (OULAD)}, Independent Research
     \href{https://avi-gobrin.github.io/personal-site/assets/oulad-sta437.pdf}{\color{blue}[Report]}}{2026}
    \cvItemListStart
      \cvItem{Ran a complete multivariate analysis pipeline on 23,301 Open University student-course records: EDA, Mahalanobis-distance MVN testing, PCA, varimax factor analysis (3 components), GMM ($K=4$), and LDA.}
      \cvItem{Showed that PCA, FA, GMM, and LDA converge on the same two-dimensional latent structure; identified three factors (Engagement, Consistent Performance, Prior Commitment) and quantified their predictive weight in a downstream classification task.}
      \cvItem{Benchmarked discriminative vs.\ generative classifiers under 5-fold CV: logistic regression 67.0\% vs.\ LDA 64.5\% vs.\ 53.0\% baseline; robustness checks isolated shared-covariance violations as the key discriminator between the two approaches.}
    \cvItemListEnd

  \cvProjectHeading
    {{\bf Autonomous Planning and Knowledge Agents}, Personal Project}{2026}
    \cvItemListStart
      \cvItem{Built a daily-planning agent (Claude Code, Obsidian) that reviews completed work, writes structured notes, and regenerates a time-blocked calendar from a parsed task file and priority scheme; reduced daily planning time from $\sim$30 minutes to under 2 minutes.}
      \cvItem{Built a school-workflow agent that auto-formats lecture and problem-set notes into a cross-linked knowledge base of theorem references, reducing note-cleanup and lookup time by an estimated 40\%.}
    \cvItemListEnd

  \cvProjectHeading
